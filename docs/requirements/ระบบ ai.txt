42. Agent Loop

หัวใจ:

while (task.isActive()) {

    context = contextEngine.build(task)

    observation = agent.observe(context)

    decision = decisionEngine.decide(
        task,
        context,
        observation
    )

    if (decision.requiresPlanning) {
        plan = planner.plan(task, context)
    }

    action = decision.action

    permission = permissionEngine.check(action)

    if (!permission.allowed) {
        requestPermission()
        continue
    }

    result = toolEngine.execute(action)

    observationEngine.record(result)

    verification = verifier.verify(
        task,
        result,
        context
    )

    if (verification.success) {
        learningEngine.learn(task)
        task.complete()
    } else {
        recoveryEngine.recover(
            task,
            verification
        )
    }
}

โค้ดจริงต้องออกแบบให้ state-safe, cancellable และ recoverable
